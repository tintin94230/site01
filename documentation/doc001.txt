supabasePays.js
import { createClient } from "@supabase/supabase-js";

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY
);

const ISO_REGEX = /^[A-Z]{2}$/;

export default async function handler(req, res) {
  try {
    // ===============================
    // GET
    // ===============================
    if (req.method === "GET") {
      const { simple, page = 0, limit = 25, search = "", sort = "nom", dir = "asc", exportCsv, used } = req.query;

      // --- Mode simple pour dropdowns (tous les pays utilisés dans transporteurs_pays) ---
      if(simple === "1" && used === "1") {
        // 1️⃣ Récupérer les IDs de pays utilisés
        const { data: tpData, error: tpError } = await supabase
          .from("transporteurs_pays")
          .select("pays_id", { distinct: true });
        if(tpError) throw tpError;

        const idsUtilises = tpData.map(t => t.pays_id);
        if(idsUtilises.length === 0) return res.json({ data: [] });

        // 2️⃣ Récupérer les pays correspondants
        const { data, error } = await supabase
          .from("pays")
          .select("id, code_iso, nom")
          .in("id", idsUtilises)
          .order("nom", { ascending: true });
        if(error) throw error;

        return res.json({ data });
      }

      // --- Pagination normale ---
      let p = parseInt(page);
      let l = parseInt(limit);
      const from = p * l;
      const to = from + l - 1;

      // -------- EXPORT CSV --------
      if (exportCsv === "1") {
        const { data, error } = await supabase
          .from("pays")
          .select("nom,code_iso,notes")
          .order("nom");
        if (error) throw error;
        const header = "nom,code_iso,notes\n";
        const rows = data.map(r =>
          `"${r.nom}","${r.code_iso}","${(r.notes ?? "").replace(/"/g,'""')}"`
        ).join("\n");
        res.setHeader("Content-Type", "text/csv");
        return res.send(header + rows);
      }

      // -------- Liste complète avec pagination et recherche --------
      let query = supabase
        .from("pays")
        .select("*", { count: "exact" })
        .order(sort, { ascending: dir === "asc" });

      if (search) query = query.ilike("nom", `%${search}%`);

      const { data, count, error } = await query.range(from, to);
      if (error) throw error;

      return res.json({ data, total: count });
    }

    // ===============================
    // POST CSV bulk
    // ===============================
    if (req.method === "POST" && req.query.importCsv === "1") {
      const rows = req.body;
      rows.forEach(r => {
        if (!ISO_REGEX.test(r.code_iso))
          throw new Error(`Code ISO invalide : ${r.code_iso}`);
      });
      const { error } = await supabase
        .from("pays")
        .upsert(rows, { onConflict: "code_iso" });
      if (error) throw error;
      return res.json({ success: true });
    }

    // ===============================
    // POST single
    // ===============================
    if (req.method === "POST") {
      let { nom, code_iso, notes } = req.body;
      code_iso = code_iso?.toUpperCase().trim();
      if (!ISO_REGEX.test(code_iso))
        return res.status(400).json({ error: "Code ISO = 2 lettres A-Z" });
      const { data, error } = await supabase
        .from("pays")
        .insert([{ nom, code_iso, notes }])
        .select();
      if (error) throw error;
      return res.json(data[0]);
    }

    // ===============================
    // PUT (mise à jour)
    // ===============================
    if (req.method === "PUT") {
      let { id, nom, code_iso, notes } = req.body;
      code_iso = code_iso?.toUpperCase().trim();
      if (!ISO_REGEX.test(code_iso))
        return res.status(400).json({ error: "Code ISO = 2 lettres A-Z" });
      const { error } = await supabase
        .from("pays")
        .update({ nom, code_iso, notes, updated_at: new Date() })
        .eq("id", id);
      if (error) throw error;
      return res.json({ success: true });
    }

    // ===============================
    // DELETE
    // ===============================
    if (req.method === "DELETE") {
      const { id } = req.body;
      await supabase.from("pays").delete().eq("id", id);
      return res.end();
    }

    return res.status(405).end();

  } catch (e) {
    console.error(e);
    res.status(500).json({ error: e.message });
  }
}

supabaseTransporteurs.js
import { createClient } from "@supabase/supabase-js";

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY
);

const URL_REGEX = /^https?:\/\/.+/i;

export default async function handler(req, res) {
  try {
    // ================= GET =================
    if (req.method === "GET") {
      const { simple, pays_id, page = 0, limit = 25, search = "", sort = "nom", dir = "asc", exportCsv } = req.query;
      const p = parseInt(page); const l = parseInt(limit);

      // --- Mode simple : dropdowns ---
      if(simple === "1" && pays_id){
        // Récupérer les IDs des transporteurs pour ce pays
        const { data: tpData, error: tpError } = await supabase
          .from("transporteurs_pays")
          .select("transporteur_id")
          .eq("pays_id", pays_id);
        if(tpError) throw tpError;

        const transporteurIds = tpData.map(tp => tp.transporteur_id);
        if(transporteurIds.length === 0) return res.json({ data: [] });

        // Récupérer les transporteurs correspondants
        const { data, error } = await supabase
          .from("transporteurs")
          .select("id, nom")
          .in("id", transporteurIds)
          .order("nom", { ascending: true });

        if(error) throw error;
        return res.json({ data });
      }

      // ---- Export CSV ----
      if (exportCsv === "1") {
        const { data, error } = await supabase
          .from("transporteurs")
          .select(`
            id, nom, type, site_web, notes,
            transporteurs_pays(pays(code_iso))
          `)
          .order("nom");
        if (error) throw error;

        const header = "nom,type,site_web,notes,pays\n";
        const rows = data.map(r => {
          const paysCodes = r.transporteurs_pays.map(tp => tp.pays.code_iso).join(";");
          return `"${r.nom}","${r.type ?? ""}","${r.site_web ?? ""}","${(r.notes ?? "").replace(/"/g,'""')}","${paysCodes}"`;
        }).join("\n");

        res.setHeader("Content-Type", "text/csv");
        return res.send(header + rows);
      }

      // ---- Liste normale avec pagination ----
      const from = p * l;
      const to = from + l - 1;

      let query = supabase
        .from("transporteurs")
        .select(`
          id, nom, type, site_web, notes,
          transporteurs_pays(pays(id,code_iso))
        `, { count: "exact" })
        .order(sort, { ascending: dir==="asc" });

      if (search) query = query.ilike("nom", `%${search}%`);

      const { data, count, error } = await query.range(from, to);
      if (error) throw error;
      return res.json({ data, total: count });
    }

    // ================= POST =================
    if (req.method === "POST") {
      const { nom, type, site_web, notes, pays } = req.body;
      if (!nom || !pays || pays.length === 0) return res.status(400).json({ error: "Nom et au moins un pays obligatoires" });
      if (site_web && !URL_REGEX.test(site_web)) return res.status(400).json({ error: "URL invalide" });

      const { data: newT, error } = await supabase
        .from("transporteurs")
        .insert([{ nom, type, site_web, notes }])
        .select();
      if (error) throw error;
      const transporteur_id = newT[0].id;

      const tpRows = pays.map(pid => ({ transporteur_id, pays_id: pid }));
      const { error: tpError } = await supabase.from("transporteurs_pays").insert(tpRows);
      if (tpError) throw tpError;

      return res.json(newT[0]);
    }

    // ================= PUT =================
    if (req.method === "PUT") {
      const { id, nom, type, site_web, notes, pays } = req.body;
      if (!id || !nom || !pays || pays.length === 0) return res.status(400).json({ error: "Nom + au moins un pays obligatoires" });
      if (site_web && !URL_REGEX.test(site_web)) return res.status(400).json({ error: "URL invalide" });

      const { error } = await supabase
        .from("transporteurs")
        .update({ nom, type, site_web, notes, updated_at: new Date() })
        .eq("id", id);
      if (error) throw error;

      await supabase.from("transporteurs_pays").delete().eq("transporteur_id", id);
      const tpRows = pays.map(pid => ({ transporteur_id: id, pays_id: pid }));
      const { error: tpError } = await supabase.from("transporteurs_pays").insert(tpRows);
      if (tpError) throw tpError;

      return res.json({ success: true });
    }

    // ================= DELETE =================
    if (req.method === "DELETE") {
      const { id } = req.body;
      if (!id) return res.status(400).json({ error: "ID manquant" });
      await supabase.from("transporteurs_pays").delete().eq("transporteur_id", id);
      await supabase.from("transporteurs").delete().eq("id", id);
      return res.end();
    }

    return res.status(405).json({ error: "Méthode non autorisée" });

  } catch(e) {
    console.error(e);
    return res.status(500).json({ error: e.message });
  }
}

import { createClient } from "@supabase/supabase-js";

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY
);

export default async function handler(req, res) {
  try {
    if (req.method === "GET") {
      const { data, error } = await supabase
        .from("beneficiaires")
        .select("*")
        .order("id", { ascending: true });

      if (error) return res.status(500).json({ error: error.message });
      return res.status(200).json({ data });
    }

    if (req.method === "POST") {
      const { code, libelle, age_max, notes } = req.body;
      if (!code || !libelle) return res.status(400).json({ error: "Champs obligatoires manquants" });

      const { data, error } = await supabase
        .from("beneficiaires")
        .insert([{ code, libelle, age_max, notes }])
        .select();

      if (error) return res.status(500).json({ error: error.message });
      return res.status(201).json(data[0]);
    }

    if (req.method === "PUT") {
      const { id, code, libelle, age_max, notes } = req.body;
      if (!id || !code || !libelle) return res.status(400).json({ error: "Champs obligatoires manquants" });

      const { data, error } = await supabase
        .from("beneficiaires")
        .update({ code, libelle, age_max, notes })
        .eq("id", id)
        .select();

      if (error) return res.status(500).json({ error: error.message });
      return res.status(200).json(data[0]);
    }

    if (req.method === "DELETE") {
      const { id } = req.body;
      if (!id) return res.status(400).json({ error: "ID manquant" });

      const { error } = await supabase
        .from("beneficiaires")
        .delete()
        .eq("id", id);

      if (error) return res.status(500).json({ error: error.message });
      return res.status(204).end();
    }

    return res.status(405).json({ error: "Méthode non autorisée" });
  } catch (err) {
    console.error("Erreur serveur:", err);
    return res.status(500).json({ error: "Erreur serveur interne" });
  }
}

import { createClient } from "@supabase/supabase-js";

const supabase = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_KEY // ← retour à ton ancienne clé
);

export default async function handler(req, res) {
  try {

    /* ================================
       GET
    ================================ */
    if (req.method === "GET") {
      const { simple, transporteur_id, pays_id, beneficiaire_id } = req.query;

      // Mode simple
      if (simple === "1") {
        let query = supabase
          .from("avantages")
          .select("id, transporteur_id, pays_id, beneficiaire_id");

        if (transporteur_id) query = query.eq("transporteur_id", transporteur_id);
        if (pays_id) query = query.eq("pays_id", pays_id);
        if (beneficiaire_id) query = query.eq("beneficiaire_id", beneficiaire_id);

        const { data, error } = await query.order("id", { ascending: true });

        if (error) return res.status(500).json({ error: error.message });
        return res.status(200).json({ data });
      }

      // GET complet avec jointures + filtres
      let query = supabase
        .from("avantages")
        .select(`
          *,
          transporteur:transporteur_id(id, nom),
          pays:pays_id(id, code_iso),
          beneficiaire:beneficiaire_id(id, code)
        `);

      if (transporteur_id) query = query.eq("transporteur_id", transporteur_id);
      if (pays_id) query = query.eq("pays_id", pays_id);
      if (beneficiaire_id) query = query.eq("beneficiaire_id", beneficiaire_id);

      const { data, error } = await query.order("id", { ascending: true });

      if (error) return res.status(500).json({ error: error.message });
      return res.status(200).json({ data });
    }

    /* ================================
       POST
    ================================ */
    if (req.method === "POST") {
      const b = req.body;

      if (!b.transporteur_id || !b.pays_id || !b.beneficiaire_id) {
        return res.status(400).json({ error: "Champs obligatoires manquants" });
      }

      const payload = {
        transporteur_id: Number(b.transporteur_id),
        pays_id: Number(b.pays_id),
        beneficiaire_id: Number(b.beneficiaire_id),

        nb_coupons_gratuits: b.nb_coupons_gratuits ?? null,
        reduction_percent: b.reduction_percent ?? null,
        prix_forfaitaire_2nde: b.prix_forfaitaire_2nde ?? null,
        prix_forfaitaire_1ere: b.prix_forfaitaire_1ere ?? null,

        conditions: b.conditions || null,
        notes: b.notes || null,
        date_debut: b.date_debut || null,
        date_fin: b.date_fin || null
      };

      const { data, error } = await supabase
        .from("avantages")
        .insert([payload])
        .select();

      if (error) return res.status(500).json({ error: error.message });
      return res.status(201).json(data[0]);
    }

    /* ================================
       PUT
    ================================ */
    if (req.method === "PUT") {
      const b = req.body;
      if (!b.id) return res.status(400).json({ error: "ID manquant" });

      const payload = {
        transporteur_id: Number(b.transporteur_id),
        pays_id: Number(b.pays_id),
        beneficiaire_id: Number(b.beneficiaire_id),

        nb_coupons_gratuits: b.nb_coupons_gratuits ?? null,
        reduction_percent: b.reduction_percent ?? null,
        prix_forfaitaire_2nde: b.prix_forfaitaire_2nde ?? null,
        prix_forfaitaire_1ere: b.prix_forfaitaire_1ere ?? null,

        conditions: b.conditions || null,
        notes: b.notes || null,
        date_debut: b.date_debut || null,
        date_fin: b.date_fin || null
      };

      const { data, error } = await supabase
        .from("avantages")
        .update(payload)
        .eq("id", b.id)
        .select();

      if (error) return res.status(500).json({ error: error.message });
      return res.status(200).json(data[0]);
    }

    /* ================================
       DELETE
    ================================ */
    if (req.method === "DELETE") {
      const { id } = req.body;
      if (!id) return res.status(400).json({ error: "ID manquant" });

      const { error } = await supabase
        .from("avantages")
        .delete()
        .eq("id", id);

      if (error) return res.status(500).json({ error: error.message });
      return res.status(204).end();
    }

    return res.status(405).json({ error: "Méthode non autorisée" });

  } catch (err) {
    console.error("Erreur serveur:", err);
    return res.status(500).json({ error: "Erreur serveur interne" });
  }
}

// =================== Save (autosave) ===================
async function save(tr){
  const id = tr.dataset.id;
  const [nom,type,site,notes] = tr.querySelectorAll("td input, td textarea");
  const pays = Array.from(tr.querySelectorAll(".pays-container select")).map(s=>parseInt(s.value,10));
  if(!nom.value || pays.length===0) return; // minimum un pays
  if(site.value && !URL.test(site.value)) return alert("URL invalide");
  await fetch("/api/supabaseTransporteurs",{
    method:"PUT",
    headers:{'Content-Type':'application/json'},
    body:JSON.stringify({ id, nom:nom.value, type:type.value, site_web:site.value, notes:notes.value, pays })
  });
}

async function addTransporteur(){

  const site=newSite.value.trim();
  const notes=newNotes.value.trim();
  const pays = Array.from(paysContainer.querySelectorAll("select")).map(s=>parseInt(s.value,10));
  if(!nom || pays.length===0) return alert("Nom et au moins un pays obligatoires");
  if(site && !/^((https?:\/\/)|(www\.))\S+\.\S+$/i.test(site)) return alert("URL invalide");
  const res = await fetch("/api/supabaseTransporteurs",{method:"POST",headers:{'Content-Type':'application/json'},body:JSON.stringify({nom,type,site_web:site,notes,pays})});

  loadTransporteurs();
}

importFile.onchange = async e=>{
  const file=e.target.files[0]; if(!file) return;
  const text = await file.text();
  const lines=text.split("\n").filter(l=>l.trim());
  const header=lines.shift().split(",");
  const rows=lines.map(l=>{
    const v=l.split(",");
    const obj={};
    header.forEach((h,i)=>obj[h]=v[i]?.replace(/"/g,''));
    if(obj.pays) obj.pays = obj.pays.split(";").map(c=>{
      const p = paysList.find(p=>p.code_iso===c);
      return p?p.id:null;
    }).filter(Boolean);
    return obj;
  });
  await fetch("/api/supabaseTransporteurs?importCsv=1",{method:"POST",headers:{'Content-Type':'application/json'},body:JSON.stringify(rows)});
  loadTransporteurs();
};

async function load(){
  const res=await fetch(`/api/supabasePays?page=${page}&limit=${limit}&search=${search}&sort=${sort}&dir=${dir}`);
  const json=await res.json();

  tbody.innerHTML='';

  json.data.forEach(p=>{

    `;

    tr.querySelectorAll('[data-field]').forEach(i=>{

    tbody.appendChild(tr);
  });

  info.textContent=`Page ${page+1}`;
}

async function autoSave(id,row){
  const inputs=row.querySelectorAll('[data-field]');
  const data={id};

  inputs.forEach(i=>data[i.dataset.field]=i.value);

  if(!ISO.test(data.code_iso)){
    row.querySelector('[data-field="code_iso"]').classList.add('invalid');
    toast('ISO invalide (2 lettres)');
    return;
  }

  await fetch('/api/supabasePays',{method:'PUT',headers:{'Content-Type':'application/json'},body:JSON.stringify(data)});
  toast('Sauvé');
}

  newNom.value=newCode.value=newNotes.value='';
  load();
}

function importCSV(input){

  <script>
    /* =====================================================
       🔑 CLÉ API OPENWEATHERMAP (version gratuite)
       ===================================================== */
    const apiKey = "e5fc1e5fa570955b439c3700e9c45d10";

    /* =====================================================
       Fonction principale : récupère et affiche la météo
       city      → nom de la ville
       targetId → id du conteneur HTML où afficher la météo
       ===================================================== */
    async function fetchWeather(city, targetId) {
      const container = document.getElementById(targetId);
      container.innerHTML = "<p>Chargement...</p>";

      try {
        /* ===== 1️⃣ Géocodage (ville → lat/lon) ===== */
        const geoResponse = await fetch(
          `https://api.openweathermap.org/geo/1.0/direct?q=${encodeURIComponent(city)}&limit=1&appid=${apiKey}`
        );

        if (!geoResponse.ok) {
          throw new Error(`Erreur géocodage (${geoResponse.status})`);
        }

        const geoData = await geoResponse.json();

        if (!geoData.length) {
          container.innerHTML = "<p>Ville non trouvée.</p>";
          return;
        }

        const { lat, lon } = geoData[0];

        /* ===== 2️⃣ Appel météo (API gratuite) ===== */
        const weatherResponse = await fetch(
          `https://api.openweathermap.org/data/2.5/weather?lat=${lat}&lon=${lon}&units=metric&lang=fr&appid=${apiKey}`
        );

        if (!weatherResponse.ok) {
          const err = await weatherResponse.json();
          throw new Error(err.message);
        }

        const data = await weatherResponse.json();

        /* ===== 3️⃣ Extraction des données utiles ===== */
        const temp = data.main.temp;
        const feels = data.main.feels_like;
        const humidity = data.main.humidity;
        const wind = data.wind.speed;
        const description = data.weather[0].description;
        const icon = data.weather[0].icon;

        /* ===== 4️⃣ Affichage ===== */
        container.innerHTML = `
          <div style="
            background:rgba(255,255,255,0.12);
            border:1px solid rgba(255,255,255,0.25);
            border-radius:16px;
            padding:20px;
            min-width:220px;
          ">
            <h3>${city}</h3>
            <img src="https://openweathermap.org/img/wn/${icon}@2x.png" alt="${description}">
            <p><strong>${temp.toFixed(1)} °C</strong> (ressenti ${feels.toFixed(1)} °C)</p>
            <p>${description}</p>
            <p>💧 Humidité : ${humidity} %</p>
            <p>💨 Vent : ${wind} m/s</p>
          </div>
        `;
      } catch (error) {
        console.error("❌ ERREUR :", error);
        container.innerHTML = `<p style="color:#ff8a80;">❌ ${error.message}</p>`;
      }
    }

    /* =====================================================
       Chargement automatique au démarrage
       ===================================================== */
    fetchWeather("Paris", "weather-paris");
    fetchWeather("Dieppe", "weather-dieppe");

    /* =====================================================
       Gestion de la ville personnalisée
       ===================================================== */
    document.getElementById("fetch-btn").addEventListener("click", () => {
      const city = document.getElementById("city-input").value.trim();
      if (!city) {
        alert("Merci de saisir une ville.");
        return;
      }
      fetchWeather(city, "weather-custom");
    });
  </script>

<h1>Page 3 – API internes & Base de données (v2)</h1>

<p>
Cette page documente les <strong>API backend internes</strong> et la <strong>structure complète de la base Supabase</strong>. 
Elle inclut les commandes pour explorer et modifier la base, et pour migrer DEV → PROD.
</p>

<hr>

<h2>1️⃣ Principe général</h2>

<p>
Toutes les pages HTML <strong>passent par des API serverless</strong> hébergées sur Vercel. 
Aucune page n’accède directement à la base de données.
</p>

<pre>
Navigateur (HTML / JS)
        |
        v
API serverless (/api/*)
        |
        v
Supabase (PostgreSQL)
</pre>

<p>
Avantages :
</p>
<ul>
  <li>Clés Supabase protégées côté serveur</li>
  <li>Centralisation de la logique métier</li>
  <li>Séparation claire Frontend / Backend</li>
  <li>Architecture prête pour évolutions futures</li>
</ul>

<hr>

<h2>2️⃣ API <code>/api/supabase001</code></h2>

<p>Utilisée par <strong>blog.html</strong> pour stocker un mot saisi par l’utilisateur.</p>

<h3>Objectif</h3>
<p>API simple et pédagogique d’interaction HTML → Supabase.</p>

<h3>Fonctions</h3>
<ul>
  <li>Recevoir un mot depuis le navigateur</li>
  <li>Calculer sa longueur</li>
  <li>Insérer le mot dans la table <code>mots</code></li>
</ul>

<h3>Endpoint</h3>
<pre>
GET /api/supabase001?mot=bonjour
</pre>

<h3>Flux</h3>
<pre>
blog.html
   |
   | fetch("/api/supabase001?mot=bonjour")
   v
API supabase001
   |
   | INSERT INTO mots
   v
Supabase
</pre>

<h3>Table <code>mots</code></h3>
<table>
  <tr><th>Colonne</th><th>Description</th></tr>
  <tr><td>id</td><td>Identifiant unique (UUID / serial)</td></tr>
  <tr><td>mot</td><td>Mot saisi</td></tr>
  <tr><td>created_at</td><td>Date de création automatique</td></tr>
</table>

<hr>

<h2>3️⃣ API <code>/api/supabase002</code></h2>

<p>Utilisée par <strong>admin.html</strong> et <strong>admin-lieux.html</strong>.</p>

<h3>Objectifs</h3>
<ul>
  <li>Gestion structurée des données</li>
  <li>Ajout / modification / suppression sécurisée</li>
  <li>Logique métier côté serveur</li>
</ul>

<h3>Données manipulées</h3>
<ul>
  <li>Lieux géographiques</li>
  <li>Itinéraires entre lieux</li>
</ul>

<h3>Flux typique</h3>
<pre>
admin-lieux.html
      |
      | fetch("/api/supabase002", data)
      v
API supabase002
      |
      | INSERT / UPDATE / SELECT / DELETE
      v
Supabase
</pre>

<h3>Sécurité</h3>
<ul>
  <li>API protégée (authentification, rôles, JWT)</li>
  <li>Aucune clé exposée côté navigateur</li>
  <li>Séparation DEV / PROD</li>
</ul>

<hr>

<h2>4️⃣ Base de données Supabase</h2>

<p>Base PostgreSQL managée par Supabase.</p>

<h3>4.1 Table <code>mots</code></h3>
<table>
  <tr><th>Colonne</th><th>Description</th></tr>
  <tr><td>id</td><td>Identifiant unique</td></tr>
  <tr><td>mot</td><td>Mot saisi</td></tr>
  <tr><td>created_at</td><td>Date de création</td></tr>
</table>

<h3>4.2 Table <code>lieux</code></h3>
<table>
  <tr><th>Colonne</th><th>Description</th></tr>
  <tr><td>id</td><td>Identifiant unique</td></tr>
  <tr><td>nom</td><td>Nom du lieu</td></tr>
  <tr><td>latitude</td><td>Coordonnée GPS</td></tr>
  <tr><td>longitude</td><td>Coordonnée GPS</td></tr>
  <tr><td>created_at</td><td>Date de création</td></tr>
</table>

<h3>4.3 Table <code>itineraires</code></h3>
<table>
  <tr><th>Colonne</th><th>Description</th></tr>
  <tr><td>id</td><td>Identifiant unique</td></tr>
  <tr><td>lieu_depart_id</td><td>Référence vers lieux.id</td></tr>
  <tr><td>lieu_arrivee_id</td><td>Référence vers lieux.id</td></tr>
  <tr><td>type_voyage</td><td>Voiture / Train / Avion…</td></tr>
  <tr><td>nom</td><td>Nom de l’itinéraire</td></tr>

  <tr><td>created_at</td><td>Date de création</td></tr>
</table>

<h4>Relations</h4>
<pre>
lieux (id)
   ▲        ▲
   |        |
itineraires
(depart) (arrivee)
</pre>

<hr>

<h2>5️⃣ Commandes utiles Supabase</h2>

<h3>5.1 Explorer la base</h3>
<pre>
supabase db remote commit
supabase db remote list
supabase db remote status
</pre>

<h3>5.2 Ajouter une colonne</h3>
<pre>
ALTER TABLE lieux ADD COLUMN description TEXT;
</pre>

<h3>5.3 Copier la structure DEV → PROD</h3>
<pre>
# Export DEV schema
supabase db dump --schema-only --file=dev_schema.sql

# Import PROD schema
supabase db restore --file=dev_schema.sql --project-ref=PROD_PROJECT_REF
</pre>

<h3>5.4 Tester les requêtes SQL</h3>
<pre>
# Exemple INSERT
INSERT INTO lieux (nom, latitude, longitude) VALUES ('Paris', 48.8566, 2.3522);

# Exemple SELECT
SELECT * FROM lieux;

# Exemple UPDATE
UPDATE lieux SET nom='Paris 1' WHERE id=1;

# Exemple DELETE
DELETE FROM lieux WHERE id=1;
</pre>

  <li>DEV → <code>.env.local</code></li>
  <li>PROD → Dashboard Vercel</li>
</ul>

<ul>

  <li>Supabase = base relationnelle PostgreSQL cloud</li>
  <li>DEV / PROD séparés, workflow sécurisé</li>

<h2>1️⃣ Visual Studio Code – Raccourcis essentiels</h2>
<ul>
  <li><strong>Ouvrir le terminal intégré :</strong> <code>CTRL + `</code></li>
  <li><strong>Formatage automatique :</strong> <code>Shift + Alt + F</code></li>
  <li><strong>Rechercher un fichier :</strong> <code>CTRL + P</code></li>
  <li><strong>Basculer entre fichiers ouverts :</strong> <code>CTRL + Tab</code></li>
  <li><strong>Fermer le terminal :</strong> <code>CTRL + Shift + `</code></li>
</ul>

<h3>Extensions utiles</h3>
<ul>
  <li>Live Server – aperçu rapide HTML</li>
  <li>Prettier – formatage code</li>
  <li>ESLint – détection erreurs JS</li>
  <li>Auto Rename Tag / HTML CSS Support</li>
</ul>

<h2>2️⃣ Commandes npm & tests locaux</h2>
<ul>
  <li><strong>Installer les dépendances :</strong> <code>npm install</code></li>
  <li><strong>Lancer le projet DEV :</strong> <code>npm run dev</code></li>
  <li><strong>Arrêter le serveur :</strong> <code>CTRL + C</code></li>
</ul>

<h3>Port 3000 occupé</h3>
<p>Si vous avez l’erreur <em>“Port 3000 already in use”</em> :</p>
<pre>
# Linux / Mac
lsof -i :3000
kill -9 PID

# Windows (PowerShell)
netstat -ano | findstr :3000
taskkill /PID PID /F
</pre>

<ul>
  <li><strong>Vérifier l’état :</strong> <code>git status</code></li>
  <li><strong>Ajouter toutes les modifications :</strong> <code>git add .</code></li>
  <li><strong>Commit :</strong> <code>git commit -m "Message clair"</code></li>
  <li><strong>Envoyer sur GitHub :</strong> <code>git push</code></li>
  <li><strong>Récupérer les mises à jour :</strong> <code>git pull</code></li>
  <li><strong>Basculer de branche :</strong> <code>git checkout nom_branche</code></li>
</ul>

<h2>6️⃣ Astuces pratiques pour reprise rapide</h2>

</ul>

Page 3 – API internes & Base de données (v2)
Cette page documente les API backend internes et la structure complète de la base Supabase. Elle inclut les commandes pour explorer et modifier la base, et pour migrer DEV → PROD.
________________________________________
1️⃣ Principe général
Toutes les pages HTML passent par des API serverless hébergées sur Vercel. Aucune page n’accède directement à la base de données.
Navigateur (HTML / JS)
        |
        v
API serverless (/api/*)
        |
        v
Supabase (PostgreSQL)
Avantages :
•	Clés Supabase protégées côté serveur
•	Centralisation de la logique métier
•	Séparation claire Frontend / Backend
•	Architecture prête pour évolutions futures
________________________________________
2️⃣ API/api/supabase001
Utilisée par blog.html pour stocker un mot saisi par l’utilisateur.
Objectif
API simple et pédagogique d’interaction HTML → Supabase.
Fonctions
•	Recevoir un mot depuis le navigateur
•	Calculer sa longueur
•	Insérer le mot dans la tablemots
Endpoint
GET /api/supabase001?mot=bonjour
Flux
blog.html
   |
   | fetch("/api/supabase001?mot=bonjour")
   v
API supabase001
   |
   | INSERT INTO mots
   v
Supabase
Tablemots
Colonne	Description
id	Identifiant unique (UUID / serial)
mot	Mot saisi
created_at	Date de création automatique
________________________________________
3️⃣ API/api/supabase002
Utilisée par admin.html et admin-lieux.html.
Objectifs
•	Gestion structurée des données
•	Ajout / modification / suppression sécurisée
•	Logique métier côté serveur
Données manipulées
•	Lieux géographiques
•	Itinéraires entre lieux
Flux typique
admin-lieux.html
      |
      | fetch("/api/supabase002", data)
      v
API supabase002
      |
      | INSERT / UPDATE / SELECT / DELETE
      v
Supabase
Sécurité
•	API protégée (authentification, rôles, JWT)
•	Aucune clé exposée côté navigateur
•	Séparation DEV / PROD
________________________________________
4️⃣ Base de données Supabase
Base PostgreSQL managée par Supabase.
4.1 Tablemots
Colonne	Description
id	Identifiant unique
mot	Mot saisi
created_at	Date de création
4.2 Tablelieux
Colonne	Description
id	Identifiant unique
nom	Nom du lieu
latitude	Coordonnée GPS
longitude	Coordonnée GPS
created_at	Date de création
4.3 Tableitineraires
Colonne	Description
id	Identifiant unique
lieu_depart_id	Référence vers lieux.id
lieu_arrivee_id	Référence vers lieux.id
type_voyage	Voiture / Train / Avion…
nom	Nom de l’itinéraire
description	Description libre
lien_http	Lien externe
created_at	Date de création
Relations
lieux (id)
   ▲        ▲
   |        |
itineraires
(depart) (arrivee)
________________________________________
5️⃣ Commandes utiles Supabase
5.1 Explorer la base
supabase db remote commit
supabase db remote list
supabase db remote status
5.2 Ajouter une colonne
ALTER TABLE lieux ADD COLUMN description TEXT;
5.3 Copier la structure DEV → PROD
# Export DEV schema
supabase db dump --schema-only --file=dev_schema.sql

# Import PROD schema
supabase db restore --file=dev_schema.sql --project-ref=PROD_PROJECT_REF
5.4 Tester les requêtes SQL
# Exemple INSERT
INSERT INTO lieux (nom, latitude, longitude) VALUES ('Paris', 48.8566, 2.3522);

# Exemple SELECT
SELECT * FROM lieux;

# Exemple UPDATE
UPDATE lieux SET nom='Paris 1' WHERE id=1;

# Exemple DELETE
DELETE FROM lieux WHERE id=1;
________________________________________
6️⃣ Environnements DEV / PROD
Variables utilisées :
SUPABASE_URL
SUPABASE_KEY
•	DEV →.env.local
•	PROD → Dashboard Vercel
7️⃣ Résumé
•	/api/supabase001 → API pédagogique mots
•	/api/supabase002 → API structurante (lieux, itinéraires)
•	Supabase = base relationnelle PostgreSQL cloud
•	DEV / PROD séparés, workflow sécurisé
•	Commandes Supabase disponibles pour exploration, modification et migration

Page 4 – Astuces & Raccourcis
Cette page compile toutes les commandes et astuces pour reprendre le projet rapidement après une pause, ou gérer des problèmes courants.
________________________________________
1️⃣ Visual Studio Code – Raccourcis essentiels
•	Ouvrir le terminal intégré :CTRL + `
•	Formatage automatique :Shift + Alt + F
•	Rechercher un fichier :CTRL + P
•	Basculer entre fichiers ouverts :CTRL + Tab
•	Fermer le terminal :CTRL + Shift + `
Extensions utiles
•	Live Server – aperçu rapide HTML
•	Prettier – formatage code
•	ESLint – détection erreurs JS
•	Auto Rename Tag / HTML CSS Support
________________________________________
2️⃣ Commandes npm & tests locaux
•	Installer les dépendances :npm install
•	Lancer le projet DEV :npm run dev
•	Arrêter le serveur :CTRL + C
Port 3000 occupé
Si vous avez l’erreur “Port 3000 already in use” :
# Linux / Mac
lsof -i :3000
kill -9 PID

# Windows (PowerShell)
netstat -ano | findstr :3000
taskkill /PID PID /F

3️⃣ Commandes Git – sauvegarde et historique

•	Ajouter toutes les modifications :git add .
•	Commit :git commit -m "Message clair"
•	Envoyer sur GitHub :git push
•	Récupérer les mises à jour :git pull
